cmake_minimum_required(VERSION 3.24)
project(qbcs_cuda LANGUAGES CXX)

option(QBCS_ENABLE_CUDA "Build the real CUDA backend (OFF builds CPU verification only)" ON)
option(QBCS_BUILD_TESTS "Build synthesis/replay correctness tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)

set(QBCS_SOURCES
  src/encoding.cpp src/synth.cpp src/algorithms_common.cpp
  src/quantum_algos.cpp src/baseline_algos.cpp src/data_loader.cpp
  src/batch_layout.cpp src/evaluation.cpp src/emulated_evaluator.cpp)

if(QBCS_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "CUDA compiler not found. Install the NVIDIA CUDA Toolkit, or configure -DQBCS_ENABLE_CUDA=OFF for CPU tests.")
  endif()
  # Ampere SASS + PTX is a portable default. For RTX 5090/5080, use
  # -DCMAKE_CUDA_ARCHITECTURES=120 with CUDA Toolkit 12.8 or newer.
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  list(APPEND QBCS_SOURCES src/cuda_evaluator.cu)
else()
  list(APPEND QBCS_SOURCES src/cuda_stub.cpp)
endif()

add_library(qbcs_core STATIC ${QBCS_SOURCES})
target_include_directories(qbcs_core PUBLIC src)
target_link_libraries(qbcs_core PUBLIC Threads::Threads)
if(QBCS_ENABLE_CUDA)
  set_target_properties(qbcs_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    CUDA_SEPARABLE_COMPILATION OFF CUDA_RUNTIME_LIBRARY Shared)
  target_link_libraries(qbcs_core PUBLIC CUDA::cudart)
  target_compile_options(qbcs_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ptxas-options=-v>)
endif()
if(MSVC)
  target_compile_options(qbcs_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/O2;/fp:precise>)
else()
  target_compile_options(qbcs_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-O2;-ffp-contract=off>)
endif()

add_executable(qbcs_cuda src/main.cpp)
target_link_libraries(qbcs_cuda PRIVATE qbcs_core)

if(QBCS_BUILD_TESTS)
  enable_testing()
  add_executable(replay_test tests/replay_test.cpp)
  target_link_libraries(replay_test PRIVATE qbcs_core)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_test.cpp")
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE qbcs_core)
    add_test(NAME integer_synthesis_equivalence COMMAND core_test)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_test.cpp")
    add_executable(service_test tests/service_test.cpp)
    target_link_libraries(service_test PRIVATE qbcs_core)
    add_test(NAME evaluation_service_concurrency COMMAND service_test)
  endif()
endif()
